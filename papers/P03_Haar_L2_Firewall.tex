\begin{abstract}
We consolidate NEU-253--258 and NEU-257 Patches 1--3.
(i)~Semiboundedness of $B_W$ on $C_c^\infty(\mathbb{R})$ relative
to the Haar-$L^2$ norm is equivalent to RH (NEU-257 \S1).
(ii)~Under RH, $B_W$ is not closable on $L^2(\mathbb{R},du)$:
its spectral measure $\mu_W=\sum_{\gamma\in\Gamma}m_\gamma\delta_{-\gamma}$
is purely atomic and singular, so the Alpay--Jorgensen criterion fails.
An explicit sequence confirms this directly.
(iii)~KLMN on $L^2(\mathbb{R},du)$ fails.
(iv)~Under RH: $\mathcal{H}_W\cong\ell^2(\Gamma,m_\gamma)$ (Suzuki 2025).
All conditional statements are RH-conditional.
$\Gamma$ is the signed zero set (Definition~\ref{def:gamma}).
Normalisation: $B_{W,\mathrm{NEU}\text{-}252}=B_{W,\mathrm{Lit}}$
(P02, Theorem~\texttt{thm:lit-id}, NEU-258 Patch~1, $\checkmark[K/M]$).
\end{abstract}

\begin{definition}[Signed zero set, NEU-257 Patch~3]\label{def:gamma}
\[
  \boxed{\Gamma := \{\gamma\in\mathbb{R} : \xi(1/2-i\gamma)=0\}}
\]
(without multiplicity). Under RH, $\Gamma\subset\mathbb{R}$ and
$-\gamma\in\Gamma$ whenever $\gamma\in\Gamma$.
Write $m_\gamma>0$ for the multiplicity of $\gamma$.
$N_\Gamma(T):=\sum_{|\gamma|\le T}m_\gamma=O(T\log T)$
(Riemann--von Mangoldt; both halves: $N_\Gamma=2N+O(1)$).
\end{definition}

\section{Precise Semiboundedness Statement}
% ---------------------------------------------------------------

\begin{theorem}[$L^2$-Semiboundedness $\Leftrightarrow$ RH,
  NEU-257 \S1 $\checkmark[K/M]$]\label{thm:semibdd}
There exists $\lambda\ge 0$ such that
$B_W(a,a)\ge-\lambda\|a\|_{L^2}^2$ for all $a\in C_c^\infty(\mathbb{R})$
if and only if RH holds.

\emph{Proof sketch.}
($\Rightarrow$) Semiboundedness $\Rightarrow$ $W+\lambda\delta_0$ positive-semidefinite
$\Rightarrow$ $W\in\mathcal{S}'$ (Bochner--Schwartz, Reed--Simon II Thm.~IX.10)
$\Rightarrow$ RH (Benedetto--Joyner; Bombieri 2000 \S3).
($\Leftarrow$) Under RH, $B_W\ge0$ (Suzuki 2011 eq.~(0.1)).
\hfill$\checkmark[K/M]$
\end{theorem}

\section{Spectral Representation under RH}
% ---------------------------------------------------------------

\begin{theorem}[Spectral representation, Suzuki 2011/2025,
  NEU-257 \S2.1 $\checkmark[K/M]$]\label{thm:spec}
Assume RH. Then
\[
  \boxed{B_W(a,b) = \sum_{\gamma\in\Gamma} m_\gamma\,
    \hat a(-\gamma)\,\overline{\hat b(-\gamma)},}
  \qquad
  \mu_W = \sum_{\gamma\in\Gamma} m_\gamma\,\delta_{-\gamma}.
\]
$\mu_W$ is purely atomic and $\mu_W\perp du$.
\end{theorem}

% ---------------------------------------------------------------
\section{Non-Closability: Alpay--Jorgensen Criterion}
% ---------------------------------------------------------------

\begin{theorem}[Non-closability, NEU-257 \S2.2
  $\checkmark[K/M]$]\label{thm:nonclosable}
Assume RH. $B_W$ is not closable on $H_0=L^2(\mathbb{R},du)$.
\begin{proof}
$B_W(a,a)=q_{\mu_W}(a)$ with $q_\sigma(f)=\int|\hat f|^2\,d\sigma$.
Alpay--Jorgensen (2012, Thm.~3.4):
$q_\sigma$ closable on $L^2(dx)\Leftrightarrow\sigma\ll dx$.
$\mu_W$ purely atomic $\Rightarrow\mu_W\perp du$
$\Rightarrow B_W$ not closable. $\square$
\end{proof}
\end{theorem}

\section{Explicit Non-Closability Sequence}
% ---------------------------------------------------------------

\begin{proposition}[Direct confirmation, NEU-257 \S2.3
  Patches~2+3 $\checkmark[K/M]$]\label{prop:explicit}
Assume RH. Fix $\gamma_0\in\Gamma$, $\gamma_0>0$,
and $\varphi\in C_c^\infty(\mathbb{R})$ with $\Phi(0):=\widehat\varphi(0)\ne0$.
Set $a_n(u):=\frac{1}{n}e^{i\gamma_0 u}\varphi(u/n)$. Then:
\begin{enumerate}
\item $\|a_n\|_2^2=\frac{1}{n}\|\varphi\|_2^2\to0$;
\item $\widehat{a_n}(-\gamma)=\Phi(n(\gamma_0-\gamma))$;
  at $\gamma_0$: $\Phi(0)\ne0$ for all $n$;
\item $B_W(a_n-a_m,a_n-a_m)=O(\min(n,m)^{-2N})$ for all $N\ge1$;
\item $B_W(a_n,a_n)\to m_{\gamma_0}|\Phi(0)|^2>0$.
\end{enumerate}
Hence $a_n\to0$ in $L^2$ and $(a_n)$ is $B_W$-Cauchy,
but $B_W(a_n,a_n)\not\to0$: $B_W$ is not closable.
\begin{proof}
(1)~Substitution $v=u/n$: $\|a_n\|_2^2=n^{-1}\|\varphi\|_2^2$.
(2)~Substitute $v=u/n$;
the $1/n$ and $n$ cancel, leaving $\Phi(n(\gamma_0-\gamma))$.
(3)~At $\gamma_0$: the difference vanishes exactly.
For $\gamma\ne\gamma_0$, $\varphi\in\mathcal{S}$ gives
$|\Phi(x)|\le C_N(1+|x|)^{-N}$, so the sum is bounded
by $C\sum_{\gamma\ne\gamma_0}m_\gamma(1+M|\gamma-\gamma_0|)^{-2N}$
($M=\min(n,m)$), which vanishes as $M\to\infty$
by Riemann--von Mangoldt summability.
(4)~The $\gamma_0$-term is constant; all others vanish. $\square$
\end{proof}
\end{proposition}

\section{Incompatibility and KLMN No-Go}
% ---------------------------------------------------------------

\begin{corollary}[Incompatibility, NEU-257 \S3
  $\checkmark[K/M]$]\label{cor:incompatible}
Semiboundedness and $L^2$-closability of $B_W$ are mutually exclusive
on $H_0=L^2(\mathbb{R},du)$.
\end{corollary}

\begin{theorem}[KLMN $\times[M]$, NEU-257 \S3
  $\checkmark[K/M]$]\label{thm:klmn}
KLMN does not apply to $B_W$ on $H_0$.
KLMN requires a closed semibounded form;
Corollary~\ref{cor:incompatible} shows this is impossible.
\end{theorem}

\section{Weil Hilbert Space under RH}
% ---------------------------------------------------------------

\begin{theorem}[Weil Hilbert space, Suzuki 2025,
  NEU-257 \S4 $\checkmark[K/M]$]\label{thm:hw}
Assume RH.
\[
  \boxed{\mathcal{H}_W \cong L^2(\tau)
  \cong \ell^2(\Gamma,m_\gamma),
  \qquad a\mapsto(\sqrt{m_\gamma}\,\hat a(-\gamma))_{\gamma\in\Gamma}.}
\]
$\mu_W\not\ll du$, so no Friedrichs extension on $L^2(du)$ exists.
This motivates P04.
\end{theorem}
