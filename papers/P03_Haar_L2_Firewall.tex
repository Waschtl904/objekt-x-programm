\begin{abstract}
We consolidate NEU-253--258 and the NEU-257 Patch.
(i)~Semiboundedness of $B_W$ on $C_c^\infty(\mathbb{R})$ relative
to the Haar-$L^2$ norm is equivalent to RH.
(ii)~Under RH, $B_W$ is not closable on $L^2(\mathbb{R},du)$,
by the criterion: a quadratic form $q_\mu$ closable on
$L^2(dx)$ if and only if $\mu\ll dx$.
(iii)~Kato--Lions--Milgram--Nelson (KLMN) on $L^2(\mathbb{R},du)$ fails.
Under RH: $\mathcal{H}_W\cong\ell^2(\Gamma,m_\gamma)$
(Suzuki 2025). All conditional statements are RH-conditional.
\end{abstract}

\section{Precise semiboundedness statement}

\begin{theorem}[$L^2$-Semiboundedness $\Leftrightarrow$ RH, NEU-257 $\checkmark$]
There exists $\lambda>0$ such that
\[
  B_W(a,a) \ge -\lambda\|a\|_{L^2(du)}^2
  \quad\forall\, a\in C_c^\infty(\mathbb{R})
\]
if and only if RH holds.

\emph{Precision:} Semiboundedness is relative to the Haar-$L^2$
norm $\|a\|_{L^2(du)}$, evaluated on the test core $C_c^\infty(\mathbb{R})$.
\end{theorem}

\section{Non-closability criterion and proof}

\begin{theorem}[Non-closability, NEU-257 $\checkmark$]
Assume RH. Then $B_W$ is not closable on $L^2(\mathbb{R},du)$.
\end{theorem}

\begin{proof}
The spectral measure of $B_W$ is
$\mu_W = \sum_\gamma m_\gamma\delta_\gamma$
(sum over imaginary parts of non-trivial zeros).
The relevant criterion is:
\[
  q_\mu\text{ closable on }L^2(dx) \iff \mu\ll dx.
\]
Since $\mu_W$ is a discrete (hence singular) measure with respect
to Lebesgue measure $du$, the condition $\mu_W\ll du$ fails.
Hence $B_W$ (as a quadratic form on $C_c^\infty(\mathbb{R})\subset L^2(du)$)
is not closable.
\end{proof}

\section{Kato--Lions--Milgram--Nelson No-Go}

\begin{theorem}[KLMN $\times[M]$, NEU-257 $\checkmark$]
The Kato--Lions--Milgram--Nelson theorem does not apply to
$B_W$ on $H_0=L^2(\mathbb{R},du)$.
\emph{KLMN = Kato--Lions--Milgram--Nelson.}
\end{theorem}

\section{Weil Hilbert space under RH}

\begin{theorem}[Weil Hilbert space, Suzuki 2025, NEU-257 $\checkmark$]
Assuming RH,
\[
  \mathcal{H}_W \cong L^2(\tau) \cong \ell^2(\Gamma,m_\gamma).
\]
\end{theorem}

\section{Consequence}

\begin{remark}
The correct Weil Hilbert geometry is singular relative to
Lebesgue measure. Specifically, $\mu_W\not\ll du$, so no
Friedrichs extension of $B_W$ on $L^2(du)$ exists. This
motivates the finite-interval construction of P04.
\end{remark}
