\section{A15.1b2c: mixed-strip descent to $R = d/2$ (full)}
\label{sec:p12-b2c}

Notation.  In this section $L$ denotes the localised-hub operator
in the three-shift chamber $\{a, b, 2a\}$ with coefficients
$(k_{a}, k_{b}, k_{2a}) = (p, r, q)$, acting as
$Lh(u) = \sum_{s} k_{s}\bigl[h(u - s) - h(u + s)\bigr]$
on the anti-reflected extension $h(-y) := -h(y)$ ($y > 0$),
with $h$ zero-extended outside $(R, S)$.  The source horizon is
$(0, T_{0})$, $T_{0} = T + \varepsilon$, $T = 2a$,
$0 < \varepsilon < \varepsilon_{\max} := \tfrac12\log(5/4)$.

Set $H(t) := h(T + t)$, $l(x) := h(T - x)$, $\Delta := p^{2} - q^{2}$.

\begin{theorem}[Mixed-strip descent to $R = d/2$]
\label{thm:p12-b2c}
Let $2a < T_{0} < c$ with $\varepsilon := T_{0} - T$,
$d/2 \le R < e$, $T < S < T_{0}$.
Then $\ker L_{R, S, T_{0}}^{\{a, b, 2a\}} = \{0\}$.
\end{theorem}

\begin{proof}[Proof of Theorem~\ref{thm:p12-b2c}]
Let $\sigma := S - T \in (0, \varepsilon)$.  Case-split on
$\sigma \le R$ and $\sigma > R$.

\smallskip
\noindent\emph{Case (i): $\sigma \le R$.}  Under $R \ge d/2 \ge e/2$,
Sub-wedge~A restricted-tail applies:
Theorem~\ref{thm:p12-b2b} gives $\ker L = \{0\}$.

\smallskip
\noindent\emph{Case (ii): $\sigma > R$.}  We use the horizon-legal
$13$-source certificate below.

\medskip
Let $h \in L^{2}(R, S)$ with $L_{T_{0}} h = 0$.  Fix
$x \in (R, \sigma)$.  We show $h(x) = 0$.

\smallskip
\noindent\emph{Step~1: horizon-legal source positions.}  Consider
\begin{align*}
u_{1} &= a + x, & u_{2} &= b - x, & u_{3} &= T + x, & u_{4} &= T - x,
& u_{5} &= d - x,\\
u_{6} &= T - d + x, & u_{7} &= 2d - x, & u_{8} &= a + b - x = T + d - x,\\
u_{9} &= e + x, & u_{10} &= T + \delta - x, & u_{11} &= T - \delta + x,\\
u_{12} &= 3d - x, & u_{13} &= 3e + x.
\end{align*}

Each $u_{i}$ lies strictly in $(0, T_{0})$:
$u_{1}, u_{2}, u_{4}, u_{5}, u_{6}, u_{7}, u_{9}, u_{10}, u_{12}, u_{13}
< T < T_{0}$ trivially (each is at most $a + b - R < T$ or a shifted
form bounded by $T$ directly);
$u_{3} = T + x < T + \sigma < T + \varepsilon = T_{0}$;
$u_{11} = T + (x - \delta) < T + \sigma < T_{0}$;
$u_{8} = T + (d - x)$, and here the horizon condition
$u_{8} < T_{0}$ reads $d - x < \varepsilon$.  This is now
\emph{automatic} in Case~(ii):
\[
R \ge d/2 \;\Longrightarrow\; d - R \le R,
\qquad
\sigma > R \text{ and } \sigma < \varepsilon
\;\Longrightarrow\; \varepsilon > \sigma > R \ge d - R.
\]
Hence $d - x < d - R \le R < \sigma < \varepsilon$ for every
$x > R$, so $u_{8} < T_{0}$.

\smallskip
\noindent\emph{Step~2: the 13 source equations, reduced by support.}
For $x \in (R, \sigma)$ with $R \ge d/2$, the operator equations at
$u_{1}, \dots, u_{13}$ reduce (after applying the anti-reflected
extension and dropping arguments outside $(R, S)$) to:
\begin{align*}
Q_{1}(u_{1}): & \quad p\, h(x) - p\, H(x) - q\, h(a - x) = 0,\\
Q_{2}(u_{2}): & \quad -p\, H(d - x) - r\, h(x) - q\, h(e + x) = 0,\\
Q_{3}(u_{3}): & \quad p\, h(a + x) + r\, h(e + x) + q\, h(x) = 0,\\
Q_{4}(u_{4}): & \quad p\, h(a - x) - q\, h(x) = 0,\\
Q_{5}(u_{5}): & \quad {-p\, h(e + x) - p\, h(b - x) - r\, h(a + x)}\\
              & \qquad {- r\, h(T + \delta - x) - q\, h(T - d + x)
                          - q\, H(d - x) = 0,}\\
Q_{6}(u_{6}): & \quad p\, h(e + x) = 0,\\
Q_{7}(u_{7}): & \quad -p\, h(T + \delta - x) - r\, h(e + x)
                       - q\, h(2e + x) = 0,\\
Q_{8}(u_{8}): & \quad p\, h(b - x) + r\, h(a - x) = 0,\\
Q_{9}(u_{9}): & \quad -p\, h(T - d + x) - r\, h(2d - x) - r\, H(x)
                       - q\, h(b - x) = 0,\\
Q_{10}(u_{10}): & \quad p\, h(2d - x) = 0,\\
Q_{11}(u_{11}): & \quad p\, h(2e + x) + r\, h(e - \delta + x) = 0,\\
Q_{12}(u_{12}): & \quad p\, h(e + 2\delta - x) - q\, h(e - \delta + x) = 0,\\
Q_{13}(u_{13}): & \quad p\, h(e - \delta + x) - q\, h(e + 2\delta - x) = 0.
\end{align*}

\smallskip
\noindent\emph{Step~3: triangular elimination.}
\begin{itemize}
\item $Q_{6}$: $h(e + x) = 0$.
\item $Q_{10}$: $h(2d - x) = 0$.
\item $Q_{12}, Q_{13}$ (with $p^{2} - q^{2} \ne 0$):
  $h(e - \delta + x) = h(e + 2\delta - x) = 0$.
\item $Q_{11}$: $h(2e + x) = 0$.
\item $Q_{7}$: $h(T + \delta - x) = 0$.
\item $Q_{4}$: $h(a - x) = (q/p)\, h(x)$.
\item $Q_{3}$: $h(a + x) = -(q/p)\, h(x)$.
\item $Q_{8}$: $h(b - x) = -(qr/p^{2})\, h(x)$.
\item $Q_{1}$: $H(x) = (\Delta / p^{2})\, h(x)$.
\item $Q_{2}$: $H(d - x) = -(r/p)\, h(x)$.
\end{itemize}

\smallskip
\noindent\emph{Step~4: closing identity from $Q_{5}$ and $Q_{9}$.}
Substituting into $Q_{5}$ gives
$h(T - d + x) = (3 r / p)\, h(x)$;
substituting into $Q_{9}$ gives
$h(T - d + x) = -r (p^{2} - 2 q^{2}) / p^{3} \cdot h(x)$.
Equating:
\[
\frac{3 r}{p}\, h(x) + \frac{r (p^{2} - 2 q^{2})}{p^{3}}\, h(x)
= \frac{2 r (2 p^{2} - q^{2})}{p^{3}}\, h(x) = 0.
\]

\smallskip
\noindent\emph{Step~5: non-vanishing.}
$p, r > 0$; $2p^{2} - q^{2} = q^{2}(4\sqrt{2} - 1) > 0$
elementary (Lemma~\ref{lem:p12-b2c-detfactors}).
Hence $h(x) = 0$ a.e. on $(R, \sigma)$.

\smallskip
\noindent\emph{Step~6: propagation.}  From Step~3 with $h(x) = 0$:
$h(a - x) = h(a + x) = h(b - x) = 0$, $H(x) = 0$, $H(d - x) = 0$
throughout $(R, \sigma)$.  Together with Case~(i)'s already-covered
regime, $h = 0$ on $(R, \sigma)$ and $H = 0$ on the tail
$(0, \sigma)$.  With $H = 0$, the $E_{\sigma}$-equation is
homogeneous on $(\sigma, a)$; the weighted-reflection kill of
Theorem~\ref{thm:p12-b2b} applies on that domain (using $R \ge d/2$
in place of $R \ge e$; only $d - x < R$ is required for
$x \in (\sigma, d)$, which holds since $d - x \le d - \sigma <
d - R \le d/2 \le R$).  This yields $h = 0$ on $(R, a)$.
$A^{+}$ at $0 < t < R$ then gives $H(t) = 0$; the tail is fully
zero, support reduces to $(R, T)$, and A15.1b1
(Theorem~\ref{thm:p12-b1}) closes the residual.
\end{proof}

\begin{corollary}[Row-multiplier certificate for Case~(ii)]
\label{cor:p12-b2c-certificate}
Under the hypotheses of Case~(ii), the following explicit
row-multiplier vector produces the closing identity directly from
$Q_{1}, \dots, Q_{13}$:
\[
(c_{1}, \dots, c_{13}) = \Bigl(
\tfrac{\Delta}{p r},\
-\tfrac{\Delta}{r^{2}},\
\tfrac{\Delta}{q r},\
-\tfrac{\Delta(p^{2} - 2q^{2})}{p^{2} q r},\
\tfrac{p \Delta}{q r^{2}},\
\tfrac{\Delta(\Delta - 2r^{2})}{p q r^{2}},\
\]
\[
-\tfrac{\Delta}{q r},\
\tfrac{\Delta^{2}}{p q r^{2}},\
-\tfrac{\Delta}{r^{2}},\
-\tfrac{\Delta}{p r},\
-\tfrac{\Delta}{p r},\
\tfrac{q}{p},\
1
\Bigr),
\]
giving
$\sum_{j = 1}^{13} c_{j}\, Q_{j}
= \bigl[2 \Delta (2p^{2} - q^{2}) / (p^{2} r)\bigr]\, h(x)$
identically.  The $13 \times 13$ coefficient matrix has determinant
$\det M_{13} = 2 p^{7} q r (p^{2} - q^{2})(2p^{2} - q^{2})$
(up to column-ordering sign), non-zero by
Lemma~\ref{lem:p12-b2c-detfactors}.
\end{corollary}

\begin{lemma}[Elementary non-vanishing]
\label{lem:p12-b2c-detfactors}
$p^{2} - q^{2} > 0$ and $2p^{2} - q^{2} > 0$ elementarily.
\end{lemma}

\begin{proof}
$p^{2} - q^{2} = q^{2}(2^{3/2} - 1) > 0$ since $2\sqrt{2} > 1$.
$2p^{2} - q^{2} = q^{2}(2 \cdot 2^{3/2} - 1) = q^{2}(4\sqrt{2} - 1)
> 0$ since $4\sqrt{2} > 1$.
\end{proof}

\subsection{b2d-upper corollary: local kill for $x > d - R$ in the b2d region}
\label{sec:p12-b2d-upper}

The 13-source certificate transfers to a sub-region of the b2d
regime.  Let $e/2 \le R < d/2$, $R < \sigma < \varepsilon <
\varepsilon_{\max}$ with $\sigma > d - R$, so that the strip
$\{d - R < x < \sigma\}$ is nonempty.  (Nonemptiness is equivalent
to $\sigma > d - R$, which requires $\varepsilon_{\max} > d - R$,
i.e.\ $R > d - \varepsilon_{\max}$; numerically $R > 0.0912$.)

\begin{corollary}[b2d-upper local kill]
\label{cor:p12-b2d-upper-local}
Under the assumptions of this subsection, for every
$x \in (d - R, \sigma)$:
$$h(x) = 0, \qquad H(x) = 0, \qquad l(x) = h(T - x) = 0.$$
Moreover $h(a - x) = h(a + x) = h(b - x) = h(2d - x) = \cdots = 0$
via the same triangular elimination chain as in
Theorem~\ref{thm:p12-b2c}, and $H(d - x) = 0$.
\end{corollary}

\begin{proof}
All $13$ source positions $u_{1}, \ldots, u_{13}$ from
Theorem~\ref{thm:p12-b2c} lie in $(0, T_{0})$ under the present
hypotheses, by the same argument as in Step~1 of the proof of
Theorem~\ref{thm:p12-b2c}, plus the following elementary
verifications for the two positions that were previously bounded
via $R \ge d/2$:
\begin{itemize}
\item $u_{8} = a + b - x = T + d - x < T + R < T + \sigma < T_{0}$,
  using $d - x < R$ (which holds since $x > d - R$).
\item $u_{12} = 3d - x < (5/2)\, d < T$, elementary via
  $3^{5} < 2^{9}$, i.e.\ $243 < 512$.
\item $u_{13} = 3e + x < 3e + \varepsilon_{\max} < T$, elementary
  via $20 < 27$ (equivalently $80 < 108$).
\end{itemize}

The four lower-support kills used in the b2c triangular elimination
transfer verbatim:
\begin{enumerate}
\item[(A)] $h(d - x) = 0$: direct from $x > d - R$.
\item[(B)] $h(e - x) = 0$: $e - x < e - R \le e/2 \le R$ (since
  $R \ge e/2$).
\item[(C)] $h(x - \delta) = 0$: $x - \delta < \varepsilon_{\max} - \delta
  < e/2 \le R$, where
  $\varepsilon_{\max} - \delta = \tfrac12 \log(10/9)$ and
  $e/2 = \tfrac14 \log(4/3)$; the inequality is elementary via
  $(10/9)^{2} < 4/3$, i.e.\ $100 < 108$.  Positivity uses
  $x > d - R > \delta$ (since $d - R > d/2 > \delta$).
\item[(D)] Where the b2c chain uses $h(2\delta - x) = 0$ or
  reflection of it: positivity $x < 2\delta$ is
  $\varepsilon_{\max} < 2\delta$, elementary via $5/4 < (9/8)^{2}$,
  i.e.\ $80 < 81$; upper bound $2\delta - x < 2\delta - (d - R)
  = R - (e - \delta) < R$.
\end{enumerate}

Since (a) all $13$ source positions are horizon-legal and (b) the
support kills (A)--(D) hold identically, the reduced source
equations $Q_{1}, \ldots, Q_{13}$ take the same form as in the proof
of Theorem~\ref{thm:p12-b2c}.  The row-multiplier certificate of
Corollary~\ref{cor:p12-b2c-certificate} then yields
$$\frac{2(p^{2} - q^{2})(2p^{2} - q^{2})}{p^{2} r}\, h(x) = 0,$$
and Lemma~\ref{lem:p12-b2c-detfactors} forces $h(x) = 0$.

The subsequent triangular back-substitution gives $H(x) = 0$ and
$H(d - x) = 0$ (via $Q_{1}$ and $Q_{2}$ respectively), and
Theorem~\ref{thm:p12-P1} at $x$ then gives
$l(x) = -c_{0}\, H(d - x) - H(x) = 0$.
\end{proof}

\begin{remark}[Scope]
\label{rem:p12-b2d-upper-scope}
Corollary~\ref{cor:p12-b2d-upper-local} is a \emph{local}
injectivity statement on the b2d-upper strip, not by itself a full b2d
result.  At the stage of this module the remaining b2d region
(b2d-core-single $R < x < d - \sigma$ and b2d-core-both
$d - \sigma < x < d - R$) was not covered by this local mechanism.
The later consolidated global result, Corollary~\ref{cor:p12-consolidated},
closes the full b2d parameter range because $e/2>\rho$.
\end{remark}
