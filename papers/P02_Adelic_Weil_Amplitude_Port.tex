\begin{abstract}
We consolidate NEU-250n--r and NEU-252 (Patch).
The port from $\mathcal{S}_{\mathrm{adel}}^{\mathrm{amp}}$
to $B_W$ proceeds in two distinct stages:
$R_{\mathrm{PW}}$ maps $F\mapsto a\in\mathcal{A}_{\mathrm{PW}}=C_c^\infty(\mathbb{R};\mathbb{C})$;
the sesquilinear evenisation $(a,b)\mapsto g_{a,b}$
gives $g_{a,b}\in\mathcal{G}_{\mathrm{ev}}^{\mathbb{C}}$.
Key properties of $g_{a,b}$: sesquilinear, even, Hermitian
($g_{b,a}=\overline{g_{a,b}}$), and $g_{a,a}$ is
\emph{positive-definite} (non-negative Fourier transform)
but \emph{not} pointwise non-negative.
All results are RH-free.\\
\textbf{Patch 3 (2026-08-08):}
Pointwise positivity $g_{a,a}\ge0$ removed ($\times[M]$,
counterexample given); Hermitian symmetry and positive-definiteness
correctly stated; $P_{\mathrm{Haar}}$ explicitly typed.
\end{abstract}

\section{Fourier Convention (NEU-220k, binding)}

\begin{definition}[Fourier convention]
\[
  \hat{f}(t) := \int_{\mathbb{R}} f(u)\,e^{itu}\,du.
\]
The $2\pi$-factor from Parseval is globally absorbed.
\end{definition}

\section{Adelic Amplitude Space (exact definition, NEU-250r)}

\begin{definition}[Haar projection, explicit]
For $F\in\mathcal{S}(\mathbb{A}_{\mathbb{Q}})$, the Haar projection onto
the archimedean component is
\[
  \boxed{
  (P_{\mathrm{Haar}}F)(x)
  := \int_{\mathbb{A}_{\mathbb{Q},f}} F(x,y)\,dy,
  \qquad \operatorname{vol}(\widehat{\mathbb{Z}})=1.
  }
\]
Here $x\in\mathbb{R}_{>0}$ and $y$ runs over the finite adeles
$\mathbb{A}_{\mathbb{Q},f}$, integrated against the Haar measure
normalised by $\operatorname{vol}(\widehat{\mathbb{Z}})=1$.
\end{definition}

\begin{definition}[Adelic amplitude space, NEU-250r $\checkmark$]
\[
  \boxed{
  \mathcal{S}_{\mathrm{adel}}^{\mathrm{amp}} :=
  \bigl\{\, F\in\mathcal{S}(\mathbb{A}_{\mathbb{Q}}) :
  (P_{\mathrm{Haar}}F)|_{(0,\infty)} \in C_c^\infty((0,\infty);\mathbb{C})
  \bigr\}.}
\]
Compact support of $(P_{\mathrm{Haar}}F)|_{(0,\infty)}$ ensures
absolute convergence of $B_{\mathrm{fin}}$.
\end{definition}

\section{Stage 1: Port Map $R_{\mathrm{PW}}$ (exact formula, NEU-250r)}

\begin{theorem}[Port map, NEU-250n--r $\checkmark$]
\[
  \boxed{
  R_{\mathrm{PW}} : \mathcal{S}_{\mathrm{adel}}^{\mathrm{amp}}
  \longrightarrow \mathcal{A}_{\mathrm{PW}},
  \qquad
  (R_{\mathrm{PW}}F)(u) := e^{u/2}(P_{\mathrm{Haar}}F)(e^u),
  \quad u\in\mathbb{R},}
\]
with $\mathcal{A}_{\mathrm{PW}}:=C_c^\infty(\mathbb{R};\mathbb{C})$.
The reparametrisation $t=\log x$ and factor $e^{u/2}$ implement the
Mellin-to-Fourier change of variables with standard Weil weight.
\end{theorem}

\section{Stage 2: Sesquilinear Evenisation}

\begin{definition}[Cross-correlation kernel]
For $a,b\in\mathcal{A}_{\mathrm{PW}}$:
\[
  C_{a,b}(u) := (a*\check{\bar{b}})(u)
  = \int_{\mathbb{R}} a(u+t)\,\overline{b(t)}\,dt.
\]
\end{definition}

\begin{definition}[Sesquilinear evenisation, NEU-250r $\checkmark$]
\[
  \boxed{
  g_{a,b}(u) := \tfrac{1}{2}\bigl[C_{a,b}(u) + C_{a,b}(-u)\bigr].}
\]
\end{definition}

\begin{proposition}[Properties of $g_{a,b}$]
\label{prop:gab}
Let $a,b\in\mathcal{A}_{\mathrm{PW}}$. Then:
\begin{enumerate}
\item \textbf{Even, compactly supported:}
  $g_{a,b}\in C_c^\infty(\mathbb{R};\mathbb{C})_{\mathrm{even}}$,
  i.e.\ $g_{a,b}\in\mathcal{G}_{\mathrm{ev}}^{\mathbb{C}}$.
\item \textbf{Sesquilinear:} $g_{a,b}$ is $\mathbb{C}$-linear in $a$
  and antilinear in $b$.
\item \textbf{Hermitian symmetry:}
  $g_{b,a}(u) = \overline{g_{a,b}(u)}$.
\item \textbf{Diagonal formula:}
  $g_{a,a}(u) = \operatorname{Re}C_{a,a}(u)\in\mathbb{R}$
  and $g_{a,a}(0)=\|a\|_{L^2}^2$.
\item \textbf{Positive-definiteness:}
  $\widehat{g_{a,a}}(t)
  =\tfrac{1}{2}\bigl(|\hat{a}(t)|^2+|\hat{a}(-t)|^2\bigr)\ge0$.
\item \textbf{Pointwise positivity fails ($\times[M]$):}
  In general $g_{a,a}(u)\not\ge0$ for all $u$
  (see counterexample below).
\end{enumerate}
\begin{proof}
(1)~Evenness: $g_{a,b}(-u)=\tfrac12[C_{a,b}(-u)+C_{a,b}(u)]=g_{a,b}(u)$.
Compact support follows from $a,b\in C_c^\infty$.\quad
(2)~Sesquilinearity: $C_{a,b}$ is linear in $a$, antilinear in $b$;
the same holds for $u\mapsto C_{a,b}(-u)$, hence for $g_{a,b}$.\quad
(3)~Hermitian symmetry: $C_{b,a}(u)=\overline{C_{a,b}(-u)}$, so
$g_{b,a}(u)=\tfrac12[C_{b,a}(u)+C_{b,a}(-u)]
=\tfrac12[\overline{C_{a,b}(-u)}+\overline{C_{a,b}(u)}]
=\overline{g_{a,b}(u)}$.\quad
(4)~Diagonal: $g_{a,a}=\tfrac12[C_{a,a}+\overline{C_{a,a}}]=\operatorname{Re}C_{a,a}$
(using $C_{a,a}(-u)=\overline{C_{a,a}(u)}$ for real $a$, and by
continuous extension for complex $a$). At $u=0$:
$C_{a,a}(0)=\int|a(t)|^2dt=\|a\|_{L^2}^2\in\mathbb{R}$.\quad
(5)~Positive-definiteness: with our Fourier convention,
$\widehat{C_{a,a}}(t)=|\hat{a}(t)|^2$ and
$\widehat{C_{a,a}(-\cdot)}(t)=|\hat{a}(-t)|^2$, hence
$\widehat{g_{a,a}}(t)=\tfrac12(|\hat{a}(t)|^2+|\hat{a}(-t)|^2)\ge0$.\quad
(6)~See counterexample.
\end{proof}
\end{proposition}

\begin{remark}[Counterexample: pointwise positivity fails, $\times[M]$]
Choose real $0\ne\varphi\in C_c^\infty(\mathbb{R})$ with support
in $[0,\delta]$, $\delta$ small, and $L>2\delta$. Set
$a(t):=\varphi(t)-\varphi(t-L)$.
The supports of $\varphi(\cdot)$ and $\varphi(\cdot-L)$ are disjoint, so
\[
  C_{a,a}(L)
  = \int a(L+t)\overline{a(t)}\,dt
  = \int\varphi(L+t)\overline{\varphi(t)}\,dt
  -\int\varphi(t)\overline{\varphi(t)}\,dt
  = 0 - \|\varphi\|_2^2.
\]
(The first integral vanishes because $\varphi(L+\cdot)$ and
$\varphi(\cdot)$ have disjoint supports.)
Since $a$ is real, $C_{a,a}(L)=C_{a,a}(-L)$, so
\[
  g_{a,a}(L)
  = \tfrac12\bigl(C_{a,a}(L)+C_{a,a}(-L)\bigr)
  = -\|\varphi\|_2^2 < 0. \qquad\times[M]
\]
Note: $\widehat{g_{a,a}}\ge0$ nevertheless (Proposition~\ref{prop:gab}(5)).
The error in an earlier draft was the inference
``non-negative Fourier transform $\Rightarrow$ pointwise non-negative'';
the correct implication is: non-negative Fourier transform
$\Leftrightarrow$ positive-definite, which does not imply
pointwise non-negativity.
\end{remark}

\section{Complete Hermitian Weil Form (NEU-252 Patch, M3)}

\begin{definition}[Weil form, M3]
\[
  B_W(a,a) = B_{\mathrm{pole}}(a,a) + B_\Gamma(a,a) + B_{\mathrm{fin}}(a,a),
\]
\[
  B_{\mathrm{fin}}(a,a) = -2\sum_{n\ge 2}\frac{\Lambda(n)}{\sqrt{n}}\,g_{a,a}(\log n),
\]
\[
  B_{\mathrm{pole}}(a,a) = \hat{g}_{a,a}(i/2)+\hat{g}_{a,a}(-i/2),
\]
\[
  B_\Gamma(a,a) = \int_{\mathbb{R}}|\hat{a}(t)|^2\,\mathrm{Re}\,\gamma_\infty(t)\,dt.
\]
\end{definition}

\begin{remark}[Sign of $B_{\mathrm{fin}}$]
Since $g_{a,a}(u)$ is not pointwise non-negative,
$B_{\mathrm{fin}}(a,a)$ does not have a definite sign from the prime-sum alone.
This is consistent with the RH-conditional nature of semiboundedness (P03).
\end{remark}

\begin{theorem}[Identification with literature, NEU-258 $\checkmark$]
$W_{\mathrm{NEU}\text{-}252} = W_{\mathrm{Lit}}$
(Weil 1952, Bombieri 2000, Suzuki 2011/2025).
\end{theorem}
