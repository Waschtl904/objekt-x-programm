\begin{abstract}
We consolidate NEU-250n--r, NEU-252 (Patch), and NEU-258 (Patch~1).
The port from $\mathcal{S}_{\mathrm{adel}}^{\mathrm{amp}}$
to $B_W$ proceeds in two distinct stages:
$R_{\mathrm{PW}}$ maps $F\mapsto a\in\mathcal{A}_{\mathrm{PW}}=C_c^\infty(\mathbb{R};\mathbb{C})$
and is surjective;
the sesquilinear evenisation $(a,b)\mapsto g_{a,b}$
gives $g_{a,b}\in\mathcal{G}_{\mathrm{ev}}^{\mathbb{C}}$.
The complete Hermitian Weil form $B_W(a,b)$ is defined
\emph{componentwise} from its three summands
$B_{\mathrm{pole}}$, $B_\Gamma$, $B_{\mathrm{fin}}$;
the polarisation identity is a derived consistency proposition.
All $2\pi$-factors are kept explicit throughout;
no global absorption convention is used.
All results are RH-free.\\
\textbf{Patch 3.2 (2026-08-08):}
Fourier convention clarified (no global $2\pi$-absorption);
$h_{a,b}$ introduced as canonical Weil test kernel;
$\gamma_\infty$ and $\Lambda_\Gamma$ given authoritative NEU-258 definitions;
$B_W$ defined componentwise;
diagonal corollary $B_\Gamma(a,a)=\frac{1}{\pi}\int|\hat{a}|^2\operatorname{Re}\gamma_\infty\,dt$
explicitly derived;
polarisation demoted from definition to Proposition~\ref{prop:polarisation}.
\end{abstract}

% ---------------------------------------------------------------
\section{Fourier Convention (NEU-220k, binding)}
% ---------------------------------------------------------------

\begin{definition}[Fourier convention]\label{def:fourier}
For $f\in L^1(\mathbb{R})$:
\[
  \hat{f}(t) := \int_{\mathbb{R}} f(u)\,e^{itu}\,du,
  \qquad
  \|\hat{f}\|_2^2 = 2\pi\,\|f\|_2^2.
\]
All $2\pi$-factors in the Weil form are kept explicitly.
No global absorption convention is used.
\end{definition}

% ---------------------------------------------------------------
\section{Adelic Amplitude Space (exact definition, NEU-250r)}
% ---------------------------------------------------------------

\begin{definition}[Haar projection, explicit]
\label{def:haar}
For $F\in\mathcal{S}(\mathbb{A}_{\mathbb{Q}})$, define
\[
  P_{\mathrm{Haar}}:
  \mathcal{S}(\mathbb{A}_{\mathbb{Q}})
  \longrightarrow
  \mathcal{S}(\mathbb{R})
\]
by
\[
  \boxed{
  (P_{\mathrm{Haar}}F)(x)
  := \int_{\mathbb{A}_{\mathbb{Q},f}} F(x,y)\,dy,
  \qquad x\in\mathbb{R},
  }
\]
where $y$ runs over the finite adeles $\mathbb{A}_{\mathbb{Q},f}$,
integrated against the Haar measure normalised by
$\operatorname{vol}(\widehat{\mathbb{Z}})=1$.
\end{definition}

\begin{definition}[Adelic amplitude space, NEU-250r $\checkmark$]
\[
  \boxed{
  \mathcal{S}_{\mathrm{adel}}^{\mathrm{amp}} :=
  \bigl\{\, F\in\mathcal{S}(\mathbb{A}_{\mathbb{Q}}) :
  (P_{\mathrm{Haar}}F)|_{(0,\infty)} \in C_c^\infty((0,\infty);\mathbb{C})
  \bigr\}.}
\]
\end{definition}

% ---------------------------------------------------------------
\section{Stage 1: Port Map $R_{\mathrm{PW}}$ (NEU-250r)}
% ---------------------------------------------------------------

\begin{theorem}[Port map, NEU-250n--r $\checkmark$]
\[
  \boxed{
  R_{\mathrm{PW}} : \mathcal{S}_{\mathrm{adel}}^{\mathrm{amp}}
  \longrightarrow \mathcal{A}_{\mathrm{PW}},
  \qquad
  (R_{\mathrm{PW}}F)(u) := e^{u/2}(P_{\mathrm{Haar}}F)(e^u),
  \quad u\in\mathbb{R}.}
\]
\end{theorem}

\begin{proposition}[Surjectivity of $R_{\mathrm{PW}}$, NEU-250r $\checkmark$]
\label{prop:rpw-surjective}
The map $R_{\mathrm{PW}}:\mathcal{S}_{\mathrm{adel}}^{\mathrm{amp}}\twoheadrightarrow C_c^\infty(\mathbb{R};\mathbb{C})$ is surjective.
For $a\in C_c^\infty(\mathbb{R};\mathbb{C})$ set
$h_a(x):=x^{-1/2}a(\log x)$ for $x>0$ and $0$ for $x\le0$,
and $F_a(x,y):=h_a(x)\,\mathbf{1}_{\widehat{\mathbb{Z}}}(y)$.
Then $F_a\in\mathcal{S}_{\mathrm{adel}}^{\mathrm{amp}}$ and $R_{\mathrm{PW}}F_a=a$. \hfill$\checkmark[K/M]$
\end{proposition}

% ---------------------------------------------------------------
\section{Stage 2: Sesquilinear Evenisation}
% ---------------------------------------------------------------

\begin{definition}[Cross-correlation kernel]
For $a,b\in\mathcal{A}_{\mathrm{PW}}$:
\[
  C_{a,b}(u) := \int_{\mathbb{R}} a(u+t)\,\overline{b(t)}\,dt.
\]
\end{definition}

\begin{definition}[Sesquilinear evenisation, NEU-250r $\checkmark$]
\[
  \boxed{g_{a,b}(u) := \tfrac{1}{2}\bigl[C_{a,b}(u) + C_{a,b}(-u)\bigr].}
\]
\end{definition}

\begin{definition}[Weil test kernel]\label{def:hab}
For $a,b\in\mathcal{A}_{\mathrm{PW}}$:
\[
  \boxed{h_{a,b}(z) := \widehat{g_{a,b}}(z)
         = \int_{\mathbb{R}} g_{a,b}(u)\,e^{izu}\,du.}
\]
Then $h_{a,b}\in\mathcal{G}_{\mathrm{ev}}^{\mathbb{C}}\cap PW$.
On $\mathbb{R}$:
\[
  h_{a,a}(t) = \tfrac{1}{2}\bigl(|\hat{a}(t)|^2 + |\hat{a}(-t)|^2\bigr).
\]
\end{definition}

\begin{proposition}[Properties of $g_{a,b}$]
\label{prop:gab}
Let $a,b\in\mathcal{A}_{\mathrm{PW}}$. Then:
\begin{enumerate}
\item \textbf{Even, compactly supported:}
  $g_{a,b}\in C_c^\infty(\mathbb{R};\mathbb{C})_{\mathrm{even}}$.
\item \textbf{Sesquilinear.}
\item \textbf{Hermitian symmetry:}
  $g_{b,a}(u) = \overline{g_{a,b}(u)}$.
\item \textbf{Diagonal formula:}
  $g_{a,a}(u) = \operatorname{Re}C_{a,a}(u)\in\mathbb{R}$;
  $g_{a,a}(0)=\|a\|_{L^2}^2$.
\item \textbf{Positive-definiteness:}
  $h_{a,a}(t) = \widehat{g_{a,a}}(t) \ge 0$.
\item \textbf{Pointwise positivity fails ($\times[M]$):}
  In general $g_{a,a}(u)\not\ge0$ pointwise.
\end{enumerate}
\begin{proof}
(1)--(5): same as Patch 3.1.  The key identity for (4) is
$C_{a,a}(-u)=\overline{C_{a,a}(u)}$ (proved by substitution $s:=t-u$),
giving $g_{a,a}=\operatorname{Re}C_{a,a}$.
(6): counterexample $a=\varphi-\varphi(\cdot-L)$ as in Patch 3.1.
\end{proof}
\end{proposition}

\section{Gamma Objects (NEU-258 Patch 1, binding)}
% ---------------------------------------------------------------

\begin{definition}[Archimedean gamma function]\label{def:gammainfty}
\[
  \boxed{\gamma_\infty(t) :=
  -\tfrac{1}{2}\log\pi
  + \tfrac{1}{2}\psi\!\left(\tfrac{1}{4}+\tfrac{it}{2}\right)}
\]
where $\psi=\Gamma'/\Gamma$ is the digamma function.
\end{definition}

\begin{definition}[Archimedean Weil functional]\label{def:lambdagamma}
\[
  \boxed{\Lambda_\Gamma(h) := \frac{1}{2\pi}\int_{\mathbb{R}}\gamma_\infty(t)\,h(t)\,dt.}
\]
\end{definition}

% ---------------------------------------------------------------
\section{Complete Hermitian Weil Form (NEU-252 Patch + NEU-258 Patch~1)}
% ---------------------------------------------------------------

\begin{definition}[Weil form, componentwise]\label{def:bw}
For $a,b\in\mathcal{A}_{\mathrm{PW}}$, define:
\begin{align}
  B_{\mathrm{pole}}(a,b) &:= h_{a,b}(i/2)+h_{a,b}(-i/2), \label{eq:bpole}\\
  B_\Gamma(a,b) &:= 2\Lambda_\Gamma(h_{a,b}), \label{eq:bgamma}\\
  B_{\mathrm{fin}}(a,b) &:= -2\sum_{n\ge2}\frac{\Lambda(n)}{\sqrt{n}}\,g_{a,b}(\log n). \label{eq:bfin}
\end{align}
The complete Hermitian Weil form is:
\[
  \boxed{B_W(a,b) := B_{\mathrm{pole}}(a,b) + B_\Gamma(a,b) + B_{\mathrm{fin}}(a,b).}
\]
\end{definition}

\begin{remark}[Prime block and symmetry, NEU-258 Patch~1]
For general $a,b$ the finite sum reads
\[
  B_{\mathrm{fin}}(a,b) = -\sum_{n\ge2}\frac{\Lambda(n)}{\sqrt{n}}\bigl[g_{a,b}(\log n)+g_{a,b}(-\log n)\bigr].
\]
For $g_{a,b}$ even (which it is by definition) this reduces to
$-2\sum_{n\ge2}\frac{\Lambda(n)}{\sqrt{n}}g_{a,b}(\log n)$.
The factor~$2$ arises from evenness; it is \emph{not} an independent
choice.
\end{remark}

\begin{corollary}[Diagonal: gamma block]\label{cor:bgamma-diag}
For $a\in\mathcal{A}_{\mathrm{PW}}$:
\begin{align}
  B_\Gamma(a,a)
  &= \frac{2}{2\pi}\int_{\mathbb{R}}\gamma_\infty(t)\cdot
     \tfrac{1}{2}\bigl(|\hat{a}(t)|^2+|\hat{a}(-t)|^2\bigr)\,dt \notag\\
  &= \frac{1}{2\pi}\int_{\mathbb{R}}\gamma_\infty(t)\,
     |\hat{a}(t)|^2\,dt
     + \frac{1}{2\pi}\int_{\mathbb{R}}\gamma_\infty(t)\,
     |\hat{a}(-t)|^2\,dt. \notag
\end{align}
Since $\operatorname{Re}\gamma_\infty$ is even
(both $\log\pi$ and $\operatorname{Re}\psi(\tfrac14+\tfrac{it}{2})$ are even in $t$),
both integrals are equal, giving:
\[
  \boxed{B_\Gamma(a,a)
  = \frac{1}{\pi}\int_{\mathbb{R}}|\hat{a}(t)|^2\,\operatorname{Re}\gamma_\infty(t)\,dt.}
\]
Equivalently:
\[
  \boxed{B_\Gamma(a,a)
  = \frac{1}{2\pi}\int_{\mathbb{R}}
    \Bigl[\operatorname{Re}\psi\!\left(\tfrac14+\tfrac{it}{2}\right)-\log\pi\Bigr]
    |\hat{a}(t)|^2\,dt.}
\]
\end{corollary}

\begin{remark}[$2\pi$ bookkeeping]
The chain is: $B_\Gamma=2\Lambda_\Gamma(h_{a,b})$, with
$\Lambda_\Gamma(h)=\frac{1}{2\pi}\int\gamma_\infty h\,dt$ and
$h_{a,a}(t)=\frac{1}{2}(|\hat a(t)|^2+|\hat a(-t)|^2)$.
The factors $2$, $\frac{1}{2\pi}$, and $\frac{1}{2}$ combine to give
the $\frac{1}{\pi}$ in Corollary~\ref{cor:bgamma-diag}.
No factor is absorbed globally.
\end{remark}

\begin{proposition}[Hermitian sesquilinearity]\label{prop:bw-sesquilinear}
$B_W$ as defined in Definition~\ref{def:bw} is Hermitian sesquilinear:
$B_W(b,a)=\overline{B_W(a,b)}$.
\begin{proof}
Each component is sesquilinear by the properties of $g_{a,b}$ and
$h_{a,b}$ (Proposition~\ref{prop:gab}, Definition~\ref{def:hab}).
\end{proof}
\end{proposition}

\begin{proposition}[Polarisation consistency]\label{prop:polarisation}
Set $Q_W(a):=B_W(a,a)$. The componentwise form agrees with
Hermitian polarisation:
\[
  B_W(a,b)
  = \frac{1}{4}\Bigl(
    Q_W(a+b)-Q_W(a-b)
    +iQ_W(a+ib)-iQ_W(a-ib)
  \Bigr).
\]
\begin{proof}
Standard: any Hermitian sesquilinear form satisfies this identity.
\end{proof}
\end{proposition}

\begin{remark}[Sign of $B_{\mathrm{fin}}$]
Since $g_{a,a}(u)$ is not pointwise non-negative,
$B_{\mathrm{fin}}(a,a)$ does not have a definite sign from the prime-sum alone.
This is consistent with the RH-conditional nature of semiboundedness (P03).
\end{remark}

\begin{theorem}[Identification with literature, NEU-258 Patch~1 $\checkmark$]
\label{thm:lit-id}
$B_{W,\mathrm{NEU\text{-}252}} = B_{W,\mathrm{Lit}}$
(Weil 1952, Bombieri 2000, Suzuki 2011/2025).
All five components are matched: Fourier convention, $\gamma_\infty$
(including $\frac{1}{2}$-factor), $\Lambda_\Gamma$ (including $\frac{1}{2\pi}$),
$B_\Gamma$ master form (including factor $2$), and $B_{\mathrm{fin}}$
(evenness of $g$).
\end{theorem}
