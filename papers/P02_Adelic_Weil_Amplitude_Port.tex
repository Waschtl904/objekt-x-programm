\begin{abstract}
We consolidate NEU-250n--r and NEU-252 (Patch).
The port from the adelic amplitude space $\mathcal{S}_{\mathrm{adel}}^{\mathrm{amp}}$
to the Weil form $B_W$ proceeds in two distinct stages.
First, $R_{\mathrm{PW}}$ maps a single adelic amplitude
$F\in\mathcal{S}_{\mathrm{adel}}^{\mathrm{amp}}$ to a compactly supported test
function $a\in\mathcal{A}_{\mathrm{PW}}=C_c^\infty(\mathbb{R};\mathbb{C})$.
Second, a pair $(a,b)$ is polarised to
$g_{a,b}\in\mathcal{G}_{\mathrm{ev}}^{\mathbb{C}}=C_c^\infty(\mathbb{R};\mathbb{C})_{\mathrm{even}}$
via the sesquilinear even\-isation $g_{a,b}(u)=\tfrac12[C_{a,b}(u)+C_{a,b}(-u)]$,
$C_{a,b}:=a*\check{\bar{b}}$.
Polarising $B_W$ yields the three-block M3 form
$B_{\mathrm{pole}}+B_\Gamma+B_{\mathrm{fin}}$.
All results are RH-free.
\textbf{Patch 2 (2026-08-08):} Exact definition of
$\mathcal{S}_{\mathrm{adel}}^{\mathrm{amp}}$ and $R_{\mathrm{PW}}$
restored from NEU-250r; sesquilinear evenisation corrected.
\end{abstract}

\section{Fourier Convention (NEU-220k, binding)}

\begin{definition}[Fourier convention]
\[
  \hat{f}(t) := \int_{\mathbb{R}} f(u)\,e^{itu}\,du.
\]
The $2\pi$-factor from Parseval is globally absorbed.
\end{definition}

\section{Adelic Amplitude Space (exact definition, NEU-250r)}

\begin{definition}[Haar projection]
Let $P_{\mathrm{Haar}}: \mathcal{S}(\mathbb{A}_{\mathbb{Q}})\to C^\infty((0,\infty);\mathbb{C})$
denote the standard Haar-measure projection of the adelic Schwartz space
onto the archimedean amplitude.
\end{definition}

\begin{definition}[Adelic amplitude space, NEU-250r $\checkmark$]
\[
  \boxed{
  \mathcal{S}_{\mathrm{adel}}^{\mathrm{amp}} :=
  \bigl\{\, F\in\mathcal{S}(\mathbb{A}_{\mathbb{Q}}) :
  (P_{\mathrm{Haar}}F)|_{(0,\infty)} \in C_c^\infty((0,\infty);\mathbb{C})
  \bigr\}.}
\]
Compact support of $(P_{\mathrm{Haar}}F)|_{(0,\infty)}$ is the
precise condition ensuring absolute convergence of the
finite prime-power sum in $B_{\mathrm{fin}}$.
\end{definition}

\section{Stage 1: Port Map $R_{\mathrm{PW}}$ (exact formula, NEU-250r)}

\begin{theorem}[Port map, NEU-250n--r $\checkmark$]
The map
\[
  \boxed{
  R_{\mathrm{PW}} : \mathcal{S}_{\mathrm{adel}}^{\mathrm{amp}}
  \longrightarrow \mathcal{A}_{\mathrm{PW}},
  \qquad
  (R_{\mathrm{PW}}F)(u) := e^{u/2}(P_{\mathrm{Haar}}F)(e^u),
  \quad u\in\mathbb{R},}
\]
is well-defined and $\mathbb{C}$-linear, with image in
$\mathcal{A}_{\mathrm{PW}}:=C_c^\infty(\mathbb{R};\mathbb{C})$.

The logarithmic reparametrisation $t=\log x$ and the factor $e^{u/2}$
exactly match the Mellin-to-Fourier change of variables and the standard
weight for the Weil functional.
\end{theorem}

\section{Stage 2: Sesquilinear Evenisation to $\mathcal{G}_{\mathrm{ev}}^{\mathbb{C}}$}

\begin{definition}[Cross-correlation kernel]
For $a,b\in\mathcal{A}_{\mathrm{PW}}$, set
\[
  C_{a,b}(u) := (a*\check{\bar{b}})(u)
  = \int_{\mathbb{R}} a(u+t)\,\overline{b(t)}\,dt, \qquad u\in\mathbb{R}.
\]
\end{definition}

\begin{definition}[Sesquilinear evenisation, NEU-250r $\checkmark$]
\[
  \boxed{
  g_{a,b}(u) := \tfrac{1}{2}\bigl[C_{a,b}(u) + C_{a,b}(-u)\bigr]
  = \tfrac{1}{2}\bigl[(a*\check{\bar{b}})(u) + (a*\check{\bar{b}})(-u)\bigr].}
\]
\end{definition}

\begin{proposition}[Properties of $g_{a,b}$, $\checkmark$]
\begin{enumerate}
\item $g_{a,b}\in C_c^\infty(\mathbb{R};\mathbb{C})_{\mathrm{even}}$, i.e.\ $g_{a,b}\in\mathcal{G}_{\mathrm{ev}}^{\mathbb{C}}$.
\item $g_{a,b}$ is sesquilinear: linear in $a$, antilinear in $b$.
\item $g_{a,a}\ge0$ on $\mathbb{R}$.
\end{enumerate}
\begin{proof}
(Geradheit) $g_{a,b}(-u)=\tfrac12[C_{a,b}(-u)+C_{a,b}(u)]=g_{a,b}(u)$.\quad
(Sesquilinearit\"at) $C_{a,b}$ is linear in $a$ and antilinear in $b$; the same
holds for $u\mapsto C_{a,b}(-u)$, hence for $g_{a,b}$.\quad
(Positivit\"at) $g_{a,a}(u)=\mathrm{Re}\,C_{a,a}(u)$
and $C_{a,a}$ is the autocorrelation of $a$, which has non-negative
Fourier transform.
\end{proof}
\end{proposition}

\begin{remark}[Type error in earlier draft corrected]
The draft contained
$g_{a,b}(u)=\tfrac12(a*\check{\bar{b}})(u)+\tfrac12(b*\check{\bar{a}})(u)$.
This interchanges $a$ and $b$ in the second term.
Since in general
$C_{b,a}(u)=\overline{C_{a,b}(-u)}\neq C_{a,b}(-u)$,
the draft formula is neither the same operation nor necessarily even.
The correct formula uses $C_{a,b}(u)+C_{a,b}(-u)$.
\end{remark}

\section{Complete Hermitian Weil Form (NEU-252 Patch, M3)}

\begin{definition}[Weil form, M3]
\[
  B_W(a,a) = B_{\mathrm{pole}}(a,a) + B_\Gamma(a,a) + B_{\mathrm{fin}}(a,a),
\]
\[
  B_{\mathrm{fin}}(a,a) = -2\sum_{n\ge 2}\frac{\Lambda(n)}{\sqrt{n}}\,g_{a,a}(\log n),
\]
\[
  B_{\mathrm{pole}}(a,a) = \hat{g}_{a,a}(i/2)+\hat{g}_{a,a}(-i/2),
\]
\[
  B_\Gamma(a,a) = \int_{\mathbb{R}}|\hat{a}(t)|^2\,\mathrm{Re}\,\gamma_\infty(t)\,dt.
\]
\end{definition}

\begin{theorem}[Identification with literature, NEU-258 $\checkmark$]
$W_{\mathrm{NEU}\text{-}252} = W_{\mathrm{Lit}}$
(Weil 1952, Bombieri 2000, Suzuki 2011/2025):
all $2\pi$-factors, the sign $-2\Lambda(n)/\sqrt{n}$,
coordinate $s=\tfrac{1}{2}+it$, and pole terms agree exactly.
\end{theorem}

\section{Canonical compression formula}

\[
  \boxed{
  \mathcal{S}_{\mathrm{adel}}^{\mathrm{amp}}
  \xrightarrow{\;F\mapsto a\;} \mathcal{A}_{\mathrm{PW}}
  \xrightarrow{\;(a,b)\mapsto g_{a,b}\;} \mathcal{G}_{\mathrm{ev}}^{\mathbb{C}}
  \xrightarrow{\mathrm{Weil}} B_W.
  }
\]

\begin{remark}[Two distinct stages]
The first arrow $R_{\mathrm{PW}}$ acts on a single amplitude
$F\mapsto a=(R_{\mathrm{PW}}F)$.
The second arrow acts on a pair $(a,b)\mapsto g_{a,b}$,
and is sesquilinear (not just linear). Merging these two stages is a type error.
\end{remark}
