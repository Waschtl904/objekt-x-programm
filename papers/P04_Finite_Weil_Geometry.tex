\begin{abstract}
We consolidate NEU-259, 260a, 260b, 260b.1.
For each $a>0$, Suzuki (2026) constructs RH-free a self-adjoint
operator $A_a$, Hilbert space $\mathcal{H}(T_a)$, and a family
$\overline{\mathscr{D}}_{a,\theta}$ of self-adjoint extensions
parametrised by $\theta\in U(1)$.

Main results (RH-free, unconditional):
\begin{enumerate}
\item Fixed local working normalization:
  $\lambda_{\mathrm{w}}(a):=\lambda_a-1$, $T_a^{\mathrm{w}}\ge I$
  (any $c>0$ works; $c=1$ is our fixed choice). This fixes each finite stage
  but does not canonise transports between different intervals.
\item Gauge freedom: $\theta\mapsto\theta+\beta-\alpha$;
  intrinsic datum is $U_a:\mathcal{N}_{+,a}\to\mathcal{N}_{-,a}$.
\item \textbf{Parity reduces $U(1)$ to $\mathbb{Z}_2$}:
  Suzuki's $PA_a=A_aP$ implies $Pv_+=v_-$, so
  $U_a\in\{+P,-P\}$, i.e.\ the parity-stable coordinates are
  $\theta(a)\in\{0,\pi\}$. Parity does not select one branch uniquely.
\end{enumerate}

Open (Objekt-X interface):
$\mathbb{Z}_2$-selection $\varepsilon(a)\in\{+1,-1\}$,
boundary normalization $\phi(a,z)$, transition maps $J_{a,b}$.
\end{abstract}

\section{Operator chain (Suzuki 2026, RH-free)}

\begin{theorem}[Operator chain, NEU-259 $\checkmark$]
For every $a>0$, unconditionally:
\begin{enumerate}
\item $Q_W^a$ is semibounded and closable on $L^2(-a,a)$.
\item $A_a=A_a^*$, discrete spectrum bounded below;
  $\lambda_a:=\inf\sigma(A_a)$ exists and is continuous in $a$.
\item For $\lambda<\lambda_a$: $T_{a,\lambda}:=A_a-\lambda I>0$
  and $\mathcal{H}(T_{a,\lambda})$ is a Hilbert space.
\item the underlying minimal symmetric operator $\mathscr{D}_a$ has deficiency
  indices $(1,1)$ on $\mathcal{H}(T_{a,\lambda})$, and each
  $\overline{\mathscr{D}}_{a,\theta}$ is a self-adjoint extension.
\end{enumerate}
\end{theorem}

\section{Fixed local working normalization (RH-free)}

\begin{proposition}[$\lambda=0$ is RH-conditional; fixed normalization, NEU-260a $\checkmark$]
For any $c>0$, $\lambda_{a,c}:=\lambda_a-c$ gives $T_{a,c}\ge cI>0$
unconditionally. We fix $c=1$ as a working normalization at each finite stage:
\[
  \lambda_{\mathrm{w}}(a) := \lambda_a-1, \qquad T_a^{\mathrm{w}} \ge I.
\]
This is a convenient local choice, not a mathematically unique or globally
transport-compatible one. The choice $\lambda=0$ requires RH and is not used
as an RH-free convention.
\end{proposition}

\section{Global shift firewall: local gauge does not canonise transport}

For $0<a<b$, let $E_{a,b}$ denote zero extension from $(-a,a)$ to $(-b,b)$.
By NEU-259 Patch~3 (using the full normalisation audit NEU-258),
$Q_W^a=B_W|_{C_c^\infty(-a,a)}$ on the test core. Hence the localised Weil
forms are compatible under zero extension:
\[
  Q_W^b(E_{a,b}v,E_{a,b}v)=Q_W^a(v,v).
\]

\begin{proposition}[Shift defect under zero extension]\label{prop:shift-defect}
For any admissible shifts $\lambda(a)<\lambda_a$ and $\lambda(b)<\lambda_b$,
\[
  \boxed{
  \|E_{a,b}v\|_{T_{b,\lambda(b)}}^2
  -
  \|v\|_{T_{a,\lambda(a)}}^2
  =
  \bigl(\lambda(a)-\lambda(b)\bigr)\|v\|_2^2.}
\]
Thus zero extension is isometric in the shifted Hilbert norms only when the
same shift is used at the two stages.
\end{proposition}

For $\lambda_{\mathrm w}(a)=\lambda_a-1$ this becomes
\[
  \boxed{
  \|E_{a,b}v\|_{T_b^{\mathrm w}}^2
  -
  \|v\|_{T_a^{\mathrm w}}^2
  =
  (\lambda_a-\lambda_b)\|v\|_2^2.}
\]
Hence the common distance~$1$ above the local spectral bottom normalises each
finite stage separately, but does not produce an isometric directed system.

\begin{proposition}[Uniform-shift firewall]\label{prop:uniform-shift}
The following are equivalent:
\begin{enumerate}
\item there exists a finite $\lambda_*$ with $\lambda_*<\lambda_a$ for every $a>0$;
\item the finite Weil forms admit one common Haar-$L^2$ lower bound on all compactly supported test functions;
\item RH holds.
\end{enumerate}
\begin{proof}
If (1) holds and $v$ is supported in $(-a,a)$, then
$Q_W^a(v,v)\ge\lambda_a\|v\|_2^2>\lambda_*\|v\|_2^2$.
Since $Q_W^a=B_W$ on that test function, this is global Haar-$L^2$
semiboundedness. By P03/NEU-257, that is equivalent to RH.
Conversely, RH gives $Q_W^a\ge0$ and hence $\lambda_a\ge0$ for every $a$,
so any fixed $\lambda_*<0$ works.
\end{proof}
\end{proposition}

\begin{remark}[Scope of the firewall]
This excludes the obvious globally isometric zero-extension system obtained
from one common shift unless RH is already true. It does not exclude a
nontrivial RH-free family of transition maps $J_{a,b}$ with stage-dependent
shifts. Proving boundedness or isometry, a cocycle law and operator
compatibility for such maps is therefore genuine additional mathematics.
\end{remark}

\section{Gauge freedom and intrinsic datum $U_a$}

\begin{proposition}[Gauge freedom, NEU-260b $\checkmark$]
The deficiency spaces are
\[
  \mathcal{N}_{\pm,a} = \operatorname{span}\{v_\pm\},
  \qquad T_a^{\mathrm{w}}\, v_\pm = e^{\pm x}.
\]
\emph{Note: $e^{\pm x}$ are on the image side of $T_a^{\mathrm{w}}$;
the deficiency vectors $v_\pm$ are their preimages in $\mathcal{H}(T_a^{\mathrm{w}})$.}

Under $v_\pm\mapsto e^{i\alpha_\pm}v_\pm$: $\theta\mapsto\theta+\beta-\alpha$.
The intrinsic datum is the unitary operator
\[
  U_a:\mathcal{N}_{+,a}\longrightarrow\mathcal{N}_{-,a}.
\]
$\theta$ is only a coordinate after a choice of bases.
\end{proposition}

\section{Parity reduces $U(1)$ to $\mathbb{Z}_2$}

\begin{theorem}[Parity theorem, NEU-260b $\checkmark$]
Let $P\,f(x):=f(-x)$ (linear, not antilinear). Suzuki (2026, \S2)
proves $PA_a=A_aP$ (the Weil kernel is even: $W^a(-x,-y)=W^a(x,y)$).

Consequences:
\begin{enumerate}
\item $PT_a^{\mathrm{w}} = T_a^{\mathrm{w}} P$, so $P$ extends
  to a unitary on $\mathcal{H}(T_a^{\mathrm{w}})$.
\item $\mathscr{D}_a P = -P\mathscr{D}_a$, so
  $P:\mathcal{N}_{+,a}\to\mathcal{N}_{-,a}$ linearly and unitarily.
\item Since $T_a^{\mathrm{w}}(Pv_+)=P(T_a^{\mathrm{w}}v_+)=Pe^x=e^{-x}=T_a^{\mathrm{w}}v_-$
  and $T_a^{\mathrm{w}}$ is injective:
  \[
    Pv_+ = v_-.
  \]
\item Parity-stable extensions require $\theta=-\theta\pmod{2\pi}$, i.e.\
  $\theta\in\{0,\pi\}$.
\end{enumerate}

\[
  \boxed{U(1) \xrightarrow{\mathrm{parity}} \{+P,-P\} \cong \mathbb{Z}_2.}
\]

In Suzuki's trivialization: $U_a^P=+P$ corresponds to $\theta=0$;
$-P$ corresponds to $\theta=\pi$.
\end{theorem}
