\subsection{Gamma anti-locality and the exact annular cancellation gate}
\label{subsec:o3ad-gamma-antilocality}

R30 reduces the fixed normal-mismatch problem to the support radius of the baseline
Riesz representative.  This subsection separates the exact archimedean Gamma rigidity
from the still-open cancellation problem for the concrete nonlocal Feshbach term.

The frozen P02 Gamma layer fixes
\[
 \gamma_\infty(\xi)
 =-\frac12\log\pi
 +\frac12\psi\!\left(\frac14+\frac{i\xi}{2}\right),
\]
so its exact even diagonal symbol is
\[
 \boxed{
 q_\Gamma(\xi)
 :=2\operatorname{Re}\gamma_\infty(\xi)
 =-\log\pi+\operatorname{Re}\psi\!\left(\frac14+\frac{i\xi}{2}\right).
 }
\tag{O3AD.1}\label{eq:o3ad-qgamma}
\]

\begin{proposition}[Exact Gamma off-support kernel and anti-locality]
\label{prop:o3ad-gamma-antilocal}
Let
\[
 \widetilde m_\Gamma(\xi)=c_0+c_1q_\Gamma(\xi),
 \qquad c_1\ne0,
\]
and let \(\widetilde M_\Gamma\) be the corresponding full-line Fourier multiplier in
the P02 convention.  Away from the diagonal its distribution kernel is
\[
 \boxed{
 \widetilde K_\Gamma(u)
 =-c_1\sum_{n=0}^\infty e^{-(2n+1/2)|u|}
 =-c_1\frac{e^{-|u|/2}}{1-e^{-2|u|}},
 \qquad u\ne0.
 }
\tag{O3AD.2}\label{eq:o3ad-gamma-kernel}
\]
If \(0\ne f\in L^2(\mathbb R)\) has
\(\operatorname{ess\,supp}f\subset[-R,R]\), then
\(\widetilde M_\Gamma f\) cannot vanish on a nonempty open interval contained in
\((R,\infty)\), and likewise on the left of \(-R\).
\end{proposition}

\begin{proof}
Put \(a=1/4\).  To avoid separating a divergent constant from the digamma series, use
the convergent identity
\[
 q_\Gamma(\xi)-q_\Gamma(0)
 =\sum_{n=0}^\infty
 \left[
 \frac1{n+a}
 -\frac{n+a}{(n+a)^2+\xi^2/4}
 \right].
\tag{O3AD.3}\label{eq:o3ad-convergent-digamma}
\]
For \(\lambda_n:=2(n+a)=2n+1/2\),
\[
 \frac{n+a}{(n+a)^2+\xi^2/4}
 =\frac{2\lambda_n}{\lambda_n^2+\xi^2}
 =\mathcal F(e^{-\lambda_n|\cdot|})(\xi).
\]
Test the inverse Fourier transform of
\eqref{eq:o3ad-convergent-digamma} against a smooth function supported away from
\(u=0\).  The constant terms contribute only distributions supported at \(0\), hence
vanish on such tests, while
\(\sum_n e^{-\lambda_n|u|}\) converges normally on every set \(|u|\ge\delta>0\).
Therefore the off-diagonal distribution kernel is exactly
\eqref{eq:o3ad-gamma-kernel}.

For \(x>R\) this gives
\[
 (\widetilde M_\Gamma f)(x)
 =-c_1\sum_{n=0}^\infty e^{-\lambda_nx}M_n(f),
 \qquad
 M_n(f):=\int_{-R}^{R}f(y)e^{\lambda_ny}\,dy.
\tag{O3AD.4}\label{eq:o3ad-moment-tail}
\]
The series converges normally for \(\operatorname{Re}x\ge R+\delta\), so the tail is
real analytic on \((R,\infty)\).  Vanishing on one open interval therefore gives
vanishing on all of \((R,\infty)\).  Multiplying successively by
\(e^{\lambda_nx}\) and sending \(x\to\infty\) yields \(M_n(f)=0\) for every \(n\).
Thus
\[
 \int_{-R}^{R}e^{y/2}f(y)e^{2ny}\,dy=0\qquad(n\ge0).
\]
With \(t=e^{2y}\), the corresponding finite complex measure on
\([e^{-2R},e^{2R}]\) annihilates every polynomial.  Polynomial density in the
continuous functions on this compact interval forces the measure, hence \(f\), to
vanish, a contradiction.  The left-tail argument is identical.
\end{proof}

\begin{remark}[Exact-symbol interface firewall]
\label{rem:o3ad-symbol-interface}
Proposition~\ref{prop:o3ad-gamma-antilocal} uses the exact frozen P02 symbol, not merely
logarithmic growth.  Equation~\eqref{eq:gamma-symbol} in P11 currently records only
\[
 m_\Gamma=1+g_\infty,
 \qquad m_\Gamma(\xi)\asymp\log(2+|\xi|).
\]
Thus the proposition transfers to the concrete P11 operator \(C_{\Gamma,S}\) only
after the affine bridge
\[
 \boxed{m_\Gamma=c_0+c_1q_\Gamma,\qquad c_1\ne0}
\tag{O3AD.5}\label{eq:o3ad-symbol-bridge}
\]
is made explicit.  Additive constants do not affect off-support kernels, but
logarithmic asymptotics alone do not determine them.
\end{remark}

For the actual fixed-\(T_0\) P11 Riesz problem put
\[
 A_X^{[T_0]}:=C_{\Gamma,X}+\Sigma_X^{[T_0]},
 \qquad
 A_X^{[T_0]}\rho_{X,T_0}=\phi_X.
\]
For \(0<R<S<T_0\), define
\[
 j_{R,S}:=E_{R,S}\rho_{R,T_0}.
\tag{O3AD.6}\label{eq:o3ad-inherited-riesz}
\]
By Proposition~\ref{prop:o3ac-riesz-equation}, applied at source level \(R\), there is
\(s_R>0\) with
\[
 E_R\rho_{R,T_0}\in H^{s_R}(\mathbb R).
\]
Since \(m_\Gamma\) has only logarithmic growth, this implies
\(m_\Gamma\widehat{E_R\rho_{R,T_0}}\in L^2(\mathbb R)\).  Hence
\[
 j_{R,S}\in\mathcal D(C_{\Gamma,S})
 =\mathcal D(A_S^{[T_0]}),
\]
so the following residual is a genuine \(L^2(-S,S)\) vector:
\[
 \boxed{
 \Delta_{R,S}^{[T_0]}
 :=\phi_S-A_S^{[T_0]}j_{R,S}.
 }
\tag{O3AD.7}\label{eq:o3ad-annular-residual}
\]

\begin{theorem}[Exact annular cancellation criterion]
\label{thm:o3ad-annular-gate}
One has
\[
 E_{R,S}^*\Delta_{R,S}^{[T_0]}=0,
\tag{O3AD.8}\label{eq:o3ad-residual-interior-zero}
\]
so the residual is supported in
\(\mathcal A_{R,S}:=(-S,-R)\cup(R,S)\).  Moreover
\[
 \boxed{
 \Delta_{R,S}^{[T_0]}=0
 \iff
 \rho_{S,T_0}=E_{R,S}\rho_{R,T_0}
 \iff
 s_{R,S,T_0}=0.
 }
\tag{O3AD.9}\label{eq:o3ad-cancellation-equivalence}
\]
Consequently, for fixed \(S<T_0\),
\[
 \boxed{
 R_*(S,T_0)=S
 \iff
 \Delta_{R,S}^{[T_0]}\ne0
 \quad\text{for every }0<R<S.
 }
\tag{O3AD.10}\label{eq:o3ad-Rstar-gate}
\]
\end{theorem}

\begin{proof}
The fixed-terminal pullback identity gives
\(E_{R,S}^*A_S^{[T_0]}E_{R,S}=A_R^{[T_0]}\) on these typed vectors, while jet
compatibility gives \(E_{R,S}^*\phi_S=\phi_R\).  Since
\(A_R^{[T_0]}\rho_{R,T_0}=\phi_R\), this proves
\eqref{eq:o3ad-residual-interior-zero}.
If \(\Delta=0\), then \(j_{R,S}\) solves the S-level Riesz equation; coercive
uniqueness gives \(j_{R,S}=\rho_{S,T_0}\).  The converse is immediate.  The equivalence
with \(s_{R,S,T_0}=0\) is Theorem~\ref{thm:o3ac-support-equivalence}, and the last
statement follows from Corollary~\ref{cor:o3ac-threshold}.
\end{proof}

\begin{corollary}[Forced annular cancellation under the exact-symbol bridge]
\label{cor:o3ad-forced-cancellation}
Assume~\eqref{eq:o3ad-symbol-bridge}.  If \(R_*(S,T_0)\le R<S\), then
\[
 \boxed{
 1_{\mathcal A_{R,S}}\Sigma_S^{[T_0]}j_{R,S}
 =1_{\mathcal A_{R,S}}\phi_S
 -1_{\mathcal A_{R,S}}C_{\Gamma,S}j_{R,S}.
 }
\tag{O3AD.11}\label{eq:o3ad-forced-cancellation}
\]
The Gamma term is real analytic on each half-annulus and has the exponential-moment
expansion~\eqref{eq:o3ad-moment-tail}; for nonzero \(j_{R,S}\), it cannot vanish on any
nonempty open subannulus.
\end{corollary}

\begin{proof}
The support assumption and Theorem~\ref{thm:o3ad-annular-gate} give \(\Delta=0\).
Splitting Gamma and Schur terms gives \eqref{eq:o3ad-forced-cancellation}; the remaining
claims follow from Proposition~\ref{prop:o3ad-gamma-antilocal}.
\end{proof}

\begin{theorem}[Coarse Schur data do not imply full support]
\label{thm:o3ad-coarse-no-go}
Fix \(0<R<S\) on the real odd subspace of \(L^2(-S,S)\).  Let \(C\ge I\) be the exact
even finite-window Gamma operator and let \(\phi=\phi_S\ne0\).  There exist
\(0\ne\rho\in C_c^\infty((-R,R))\) and a bounded positive selfadjoint rank-one
operator \(\Sigma\ge0\), preserving parity and bounded on a symmetric sufficiently
small Sobolev interval, such that
\[
 \boxed{(C+\Sigma)\rho=\phi.}
\tag{O3AD.12}\label{eq:o3ad-abstract-countermodel}
\]
Moreover \(\Sigma=H_0B_0H_0^*\) with
\(B_0=(I+0^*0)^{-1}=I\).
\end{theorem}

\begin{proof}
Choose real odd \(h\in C_c^\infty((-R,R))\) with
\(\langle h,\phi\rangle>0\).  Set \(\rho=\varepsilon h\) and
\(d=\phi-C\rho\).  For small \(\varepsilon>0\),
\[
 \langle\rho,d\rangle
 =\varepsilon\langle h,\phi\rangle
 -\varepsilon^2\langle h,Ch\rangle>0.
\]
Put
\[
 e=\rho/\|\rho\|,
 \quad y=d/\|\rho\|,
 \quad a=\langle e,y\rangle>0,
 \quad z=y-ae\perp e,
\]
and
\[
 v=\sqrt a\,e+a^{-1/2}z,
 \qquad \Sigma f=\langle f,v\rangle v.
\]
Then \(\Sigma\ge0\), is selfadjoint and rank one, and
\(\Sigma e=\sqrt a\,v=y\); hence \(\Sigma\rho=d\).  All vectors are odd, so parity is
preserved.  Lemma~\ref{lem:gamma-smooth-action} and the same \(|s|<1/2\) cutoff range
used in O3k give the required small-Sobolev boundedness of this rank-one map.  Finally
take \(H_0=\Sigma^{1/2}\), \(B_0=I\).
\end{proof}

\begin{remark}[Scope and finite-propagation firewall]
\label{rem:o3ad-countermodel-scope}
Theorem~\ref{thm:o3ad-coarse-no-go} is not a counterexample to the concrete P11
operator
\[
 \Sigma_S^{[T_0]}
 =E_{S,T_0}^*H_{T_0}(I+R_{T_0}^*R_{T_0})^{-1}H_{T_0}^*E_{S,T_0}.
\]
It rules out only the proof route using Gamma anti-locality plus positivity,
boundedness, parity, small-Sobolev regularity and an abstract positive Feshbach
factorisation.  Although \(H_{T_0}\) and the resolved \(R_{T_0}\) are finite sums of
translations and interval cutoffs, O3k proves no finite propagation or support
preservation for \(B_{T_0}=(I+R_{T_0}^*R_{T_0})^{-1}\).  Such a property cannot be
inserted without a separate proof.
\end{remark}

\begin{openproblem}[R31: concrete annular noncancellation]
\label{open:o3ad-concrete-cancellation}
For every fixed \(0<R<S<T_0\), prove or disprove
\[
 \boxed{\Delta_{R,S}^{[T_0]}\ne0.}
\]
By Theorem~\ref{thm:o3ad-annular-gate}, this family is exactly equivalent to R30-F,
\(R_*(S,T_0)=S\).  Under the exact-symbol bridge, a positive proof must rule out the
concrete identity~\eqref{eq:o3ad-forced-cancellation}; Gamma anti-locality alone is
insufficient.  No conclusion here concerns the relative polar gauge or strong terminal
transport.
\end{openproblem}
