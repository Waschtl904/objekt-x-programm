#!/usr/bin/env python3
"""Diagnose the A1 C-odd builder/evaluation import-state mismatch.

This is an audit regression, NOT an operator-positivity certificate.
It imports the unchanged frozen sources in fresh subprocesses and compares
initialised constants, unquantised proposals, and four sampled dyadic rows.
No full matrix, full node sweep, or LDL computation is performed.
"""
import argparse
import hashlib
import importlib
import json
import os
from pathlib import Path
import subprocess
import sys

SOURCE_COMMIT = "ac164bbbd2c46623aa64e567d21f813f41f164b0"
SOURCES = {
    "check_a1_c_even_engine_preflight_arb.py": "a3d751ac41255a2f2f450ba29432d83e56bc748d",
    "certify_a1_c_even_exact_dyadic_matrix.py": "84630733b352e572385414351013d0fc6832dda3",
    "certify_a1_c_odd_exact_dyadic_matrix.py": "655b9240f4ab77c4e0228fd7621e5c89bc7a2fc3",
    "a1_c_odd_matrix_artifact.py": "f6f2d151c6c4b0143a71b778a8fd6a9327ce47fa",
    "check_a1_c_odd_eval_shard_arb.py": "8deefeaf6c17628fc436b0cf9010960922540e5c",
}
SAMPLES = ((0, 0), (1250, 20), (2500, 20), (3877, 39))


def source_check(root):
    for name, expected in SOURCES.items():
        raw = (root / name).read_bytes()
        actual = hashlib.sha1(b"blob " + str(len(raw)).encode() + b"\0" + raw).hexdigest()
        if actual != expected:
            raise RuntimeError(f"source drift: {name}: {actual} != {expected}")


def exact_point(value):
    return [int(x) for x in value.man_exp()]


def digest(value):
    return hashlib.sha256(json.dumps(value, separators=(",", ":")).encode()).hexdigest()


def worker(case, root):
    # Reproduce the historical workflow: no A1_PREC_BITS before builder import.
    os.environ.pop("A1_PREC_BITS", None)
    os.environ["A1_SHARD_INDEX"] = "0"
    os.environ["A1_SHARD_COUNT"] = "32"
    sys.path.insert(0, str(root))
    if case == "builder":
        importlib.import_module("a1_c_odd_matrix_artifact")
    elif case == "shard":
        importlib.import_module("check_a1_c_odd_eval_shard_arb")
    else:
        raise ValueError(case)
    engine = importlib.import_module("check_a1_c_even_engine_preflight_arb")
    builder = importlib.import_module("certify_a1_c_odd_exact_dyadic_matrix")
    # Importing the validator now does NOT recreate the cached engine constants.
    validator = importlib.import_module("check_a1_c_odd_eval_shard_arb")
    from flint import arb, ctx
    if ctx.prec != 3072:
        raise RuntimeError(f"unexpected final working precision: {ctx.prec}")
    result = {
        "case": case,
        "final_ctx_prec": ctx.prec,
        "pi_mid_bits": engine.PI.bits(),
        "pi_mid_digest": digest(exact_point(engine.PI.mid())),
        "pi_radius_binary": exact_point(engine.PI.rad()),
        "prime_constants_digest": digest([
            [n, exact_point(logn.mid()), exact_point(amp.mid())]
            for n, logn, amp in engine.PRIME_DATA
        ]),
        "samples": [],
    }

        x, w = engine.gauss_node(panel, node)
        head = builder.sharp_odd_head(x)

        alpha_q, b_q = builder.quantized_node(panel, node)
        eb = validator.vector_error_bound(x, head)
        result["samples"].append({
            "panel": panel, "node": node,
            "alpha_mid_digest": digest(exact_point(alpha.mid())),
            "bessel_mid_digest": digest([[n, exact_point(v.mid())] for n, v in head]),
            "dyadic_row_digest": digest([alpha_q, b_q]),
            "same_state_e_b_bound": eb.str(30),
            "same_state_e_b_below_1e_43": bool(eb < arb("1e-43")),
        })
    print(json.dumps(result, indent=2))


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--worker", choices=("builder", "shard"))
    parser.add_argument("--output", default="a1_import_state_diagnostic.json")
    args = parser.parse_args()
    root = Path(__file__).resolve().parent
    source_check(root)
    if args.worker:
        worker(args.worker, root)
        return
    states = {}
    for case in ("builder", "shard"):
        proc = subprocess.run(
            [sys.executable, str(Path(__file__).resolve()), "--worker", case],
            check=True, capture_output=True, text=True, timeout=120,
        )
        states[case] = json.loads(proc.stdout)
    b, s = states["builder"], states["shard"]
    if not (b["pi_mid_bits"] <= 512 and s["pi_mid_bits"] > 3000):
        raise RuntimeError("historical initialisation mismatch was not reproduced")
    if b["pi_mid_digest"] == s["pi_mid_digest"]:
        raise RuntimeError("unexpected identical pi midpoints")
    if b["prime_constants_digest"] == s["prime_constants_digest"]:
        raise RuntimeError("unexpected identical prime-constant proposals")

    for br, sr in zip(b["samples"], s["samples"]):
        comparisons.append({
            "panel": br["panel"], "node": br["node"],
            "alpha_midpoints_equal": br["alpha_mid_digest"] == sr["alpha_mid_digest"],
            "bessel_midpoints_equal": br["bessel_mid_digest"] == sr["bessel_mid_digest"],
            "dyadic_rows_equal": br["dyadic_row_digest"] == sr["dyadic_row_digest"],

        "sample_comparisons": comparisons,
        "firewall": "Four samples are not all-node validation or proof of whole-matrix equality. No C-odd transfer or full A1 promotion.",
    }
    Path(args.output).write_text(json.dumps(report, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(report, indent=2))


if __name__ == "__main__":
    main()
