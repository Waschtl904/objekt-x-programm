"""Rigorous Arb certificate for a sharpened A1 Legendre tail using DLMF 10.14.5.

This checker does NOT certify the final finite matrices. It replaces the crude
power/double-factorial band-tail bound from PR #117 by the much sharper DLMF
large-order inequality

    |J_nu(nu*x)| <= x^nu exp(nu*sqrt(1-x^2)) / (1+sqrt(1-x^2))^nu,
    nu >= 0, 0 < x <= 1.

For spherical j_n(t)=sqrt(pi/(2t)) J_{n+1/2}(t), write

    nu = n + 1/2,
    phi(x) = log x + sqrt(1-x^2) - log(1+sqrt(1-x^2)).

Then phi'(x)=sqrt(1-x^2)/x.  The DLMF envelope

    H_nu(t) = t^(-1) exp(2 nu phi(t/nu))

satisfies

    d/dt log H_nu(t) = (2 nu sqrt(1-(t/nu)^2)-1)/t.

Hence H_nu is increasing on [0,Omega] whenever n(n+1)>Omega^2.
This gives the concentration diagonal bound

    d_n <= 2 nu exp(2 nu phi(Omega/nu)).

Let F(nu)=log(2nu)+2nu phi(Omega/nu).  Since

    F'(nu)=1/nu + 2 log(x/(1+s)),
    F''(nu)=-1/nu^2-2/(nu*s) < 0,

we get

    d_{n+1}/d_n <= exp(F'(nu))
                 = exp(1/nu) (x/(1+s))^2,

and the right side decreases with n. Therefore the whole Legendre band tail is
geometrically majorized from the first discarded degree M onward.

Predeclared gate:

    Omega = 1551,
    M = 1730,
    parity head dimensions = 865 + 865.

No binary float enters an acceptance test.
"""

from flint import arb, ctx

ctx.prec = 256

PI = arb.pi()
OMEGA = arb(1551)
C = arb("0.1")
R = arb(12)
M = 1730
ETA_TARGET = arb("1.2e-46")
PENALTY_TARGET = arb("2e-43")
FINITE_TARGET = arb("1e-35")


def odd_double_factorial(n: int) -> int:
    assert n >= 1 and n % 2 == 1
    out = 1
    for k in range(1, n + 1, 2):
        out *= k
    return out


def dlmf_band_tail_bound() -> tuple[arb, arb, arb]:
    """Return (eta_M, d_M, q_M) from the DLMF 10.14.5 envelope."""
    # Certify monotonicity condition n(n+1)>Omega^2 at the first discarded n.
    if not (arb(M * (M + 1)) > OMEGA * OMEGA):
        raise RuntimeError("DLMF spherical envelope is not monotone on the full band")

    nu = arb(M) + arb(1) / 2
    x = OMEGA / nu
    if not (x > 0 and x < 1):
        raise RuntimeError(f"invalid DLMF scaled argument x={x}")

    s = (1 - x * x).sqrt()
    phi = x.log() + s - (1 + s).log()

    # Diagonal concentration bound at n=M.
    dM = 2 * nu * (2 * nu * phi).exp()

    # Concavity-derived ratio bound for all later n.
    q = (1 / nu).exp() * (x / (1 + s)) ** 2
    if not (q < 1):
        raise RuntimeError(f"DLMF tail ratio bound is not <1: q={q}")

    eta = dM / (1 - q)
    return eta, dM, q


def moment_tail_bound_one_weight() -> arb:
    """Existing rigorous tail bound for the Legendre coefficients of e^{x/2}."""
    z = arb(1) / 2
    df = arb(odd_double_factorial(2 * M + 1))
    nu = arb(M) + arb(1) / 2
    exponent = (z * z) / arb(2 * M + 3)
    sM = 4 * nu * (z ** (2 * M)) / (df * df) * exponent.exp()

    q = ((arb(M) + arb(3) / 2) / (arb(M) + arb(1) / 2))
    q *= (z * z) / arb((2 * M + 3) ** 2)
    if not (q < 1):
        raise RuntimeError(f"moment tail ratio bound is not <1: q={q}")
    return sM / (1 - q)


def main() -> None:

        raise RuntimeError(f"DLMF Legendre band tail too large: eta={eta}")

    delta_E = (2 * sigma2).sqrt()

    # ||E|| <= sqrt(||e^{x/2}||^2+||e^{-x/2}||^2)=sqrt(4 sinh 1).

        raise RuntimeError(f"sharpened Legendre tail floor too small: tau={tau}")
