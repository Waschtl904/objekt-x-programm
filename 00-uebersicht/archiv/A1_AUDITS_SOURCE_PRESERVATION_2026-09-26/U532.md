#!/usr/bin/env python3
"""Exact equality bridge for historical and shard-initialised C-odd data.

Two fresh spawned processes import the unchanged frozen sources in different
orders. EVERY scalar/row integer and all 1075 moment integers are compared
as Python integers, not via hash equality. No Gram matrix or LDL is built.
The historical artifact identity is imported from its recorded source build;
the canonical midpoint error bounds remain the earlier shard certificates.
"""
import argparse
import hashlib
import importlib
import json
import multiprocessing as mp
import os
from pathlib import Path
import sys

from check_a1_composition_import_state import SOURCE_COMMIT, source_check

PANELS = 3878
NODES = 40
SHARDS = 32
DIM = 1075


def worker(case, shard, root, sender):
    try:
        os.environ.pop("A1_PREC_BITS", None)
        os.environ["A1_SHARD_COUNT"] = str(SHARDS)
        os.environ["A1_SHARD_INDEX"] = str(shard)
        sys.path.insert(0, root)
        if case == "historical":
            importlib.import_module("a1_c_odd_matrix_artifact")
        elif case == "canonical":
            importlib.import_module("check_a1_c_odd_eval_shard_arb")
        else:
            raise ValueError(case)
        builder = importlib.import_module("certify_a1_c_odd_exact_dyadic_matrix")
        engine = importlib.import_module("check_a1_c_even_engine_preflight_arb")
        from flint import ctx
        sender.send({"kind": "initialisation", "case": case,
                     "ctx_prec": ctx.prec, "pi_bits": engine.PI.bits()})
        count = 0
        for panel in range(shard, PANELS, SHARDS):
            for node in range(NODES):
                alpha, row = builder.quantized_node(panel, node)
                if len(row) != DIM:
                    raise RuntimeError("wrong row dimension")
                sender.send({"kind": "row", "panel": panel, "node": node,
                             "alpha": alpha, "row": row})
                count += 1
        moments = builder.quantized_moment_vector()
        sender.send({"kind": "moments", "values": moments})
        sender.send({"kind": "done", "nodes": count})
    except BaseException as exc:
        sender.send({"kind": "error", "detail": repr(exc)})
        raise
    finally:
        sender.close()


def receive(reader):
    if not reader.poll(180):
        raise RuntimeError("worker communication timeout")
    message = reader.recv()
    if message.get("kind") == "error":
        raise RuntimeError(message["detail"])
    return message


def compare_shard(shard):
    root = Path(__file__).resolve().parent
    source_check(root)
    context = mp.get_context("spawn")
    processes, readers = [], []
    for case in ("historical", "canonical"):
        reader, writer = context.Pipe(duplex=False)
        process = context.Process(target=worker, args=(case, shard, str(root), writer))
        process.start()
        writer.close()
        processes.append(process)
        readers.append(reader)
    row_digest = hashlib.sha256()
    count = 0
    try:
        historical, canonical = [receive(r) for r in readers]
        if (historical["kind"] != "initialisation" or canonical["kind"] != "initialisation"
                or historical["ctx_prec"] != 3072 or canonical["ctx_prec"] != 3072
                or not historical["pi_bits"] <= 512 or not canonical["pi_bits"] > 3000):
            raise RuntimeError("unexpected initialisation states")
        for panel in range(shard, PANELS, SHARDS):
            for node in range(NODES):
                left, right = [receive(r) for r in readers]
                for item in (left, right):
                    if item.get("kind") != "row" or item.get("panel") != panel or item.get("node") != node:
                        raise RuntimeError("row order/coverage mismatch")
                    if len(item["row"]) != DIM:
                        raise RuntimeError("wrong dimension")
                # Exact equality is the acceptance test. Digests below are provenance only.
                if left["alpha"] != right["alpha"] or left["row"] != right["row"]:
                    raise RuntimeError(f"DYADIC_PROPOSAL_MISMATCH panel={panel} node={node}")
                row_digest.update(json.dumps(left, sort_keys=True, separators=(",", ":")).encode())
                row_digest.update(b"\n")
                count += 1
        left, right = [receive(r) for r in readers]
        if (left.get("kind") != "moments" or right.get("kind") != "moments"
                or len(left["values"]) != DIM or left["values"] != right["values"]):
            raise RuntimeError("moment proposal mismatch")
        moment_digest = hashlib.sha256(json.dumps(left["values"], separators=(",", ":")).encode()).hexdigest()
        final_left, final_right = [receive(r) for r in readers]
        expected = len(range(shard, PANELS, SHARDS)) * NODES
        if (final_left != {"kind": "done", "nodes": expected}
                or final_right != {"kind": "done", "nodes": expected} or count != expected):
            raise RuntimeError("incomplete node coverage")
        for process in processes:
            process.join(10)
            if process.exitcode != 0:
                raise RuntimeError(f"worker exit code {process.exitcode}")
        return {"source_commit": SOURCE_COMMIT, "shard": shard, "shards": SHARDS,
                "nodes_compared": count, "moment_integers_compared": DIM,
                "historical_pi_bits": historical["pi_bits"],
                "canonical_pi_bits": canonical["pi_bits"],
                "row_stream_sha256": row_digest.hexdigest(), "moment_sha256": moment_digest,
                "acceptance": "EXACT_INTEGER_EQUALITY", "status": "PASS",
                "firewall": "Identity bridge only. Imports the canonical evaluation budgets and the historical artifact build; no new SPD replay or full A1 promotion."}
    finally:
        for reader in readers:
            reader.close()
        for process in processes:
            if process.is_alive():
                process.terminate()
            process.join(10)


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--shard", type=int, required=True)
    args = parser.parse_args()
    if not 0 <= args.shard < SHARDS:
        raise ValueError("invalid frozen shard index")
    report = compare_shard(args.shard)
    output = Path(f"a1_odd_proposal_identity_{args.shard}.json")
    output.write_text(json.dumps(report, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(report, indent=2))
    print("CERTIFIED: historical and canonical C-odd dyadic proposals are exactly equal on this shard")


if __name__ == "__main__":
    main()
