"""Rational/scalar regressions only; no continuum or full-window certificate.

No imports from A1/C0 or old checkers; no files/network/CI are touched.
Analytic input: the source note's independent small-support floors.

from fractions import Fraction as F
import json

COUNT = 0

def check(ok, label):
    global COUNT
    if not ok:
        raise ArithmeticError('FAIL: ' + label)
    COUNT += 1
    print('PASS: ' + label)

def tr(a):
    return [list(row) for row in zip(*a)]

def mul(a, b):
    if len(a[0]) != len(b):
        raise ValueError('incompatible matrix sizes')
    return [[sum((x*y for x,y in zip(row,col)),F(0)) for col in zip(*b)] for row in a]

def add(a, b, sign=1):
    return [[x+sign*y for x,y in zip(ar,br)] for ar,br in zip(a,b)]

def inverse(a):
    n = len(a)
    z = [list(map(F,row))+[F(i==j) for j in range(n)] for i,row in enumerate(a)]
    for j in range(n):
        i = next((i for i in range(j,n) if z[i][j]), None)
        if i is None:
            raise ValueError('singular matrix')
        z[j],z[i] = z[i],z[j]
        p = z[j][j]
        z[j] = [v/p for v in z[j]]
        for i in range(n):
            if i != j:
                p=z[i][j]
                z[i]=[u-p*v for u,v in zip(z[i],z[j])]
    return [r[n:] for r in z]

def scalar(a):
    if len(a)!=1 or len(a[0])!=1:
        raise ValueError('not scalar')
    return a[0][0]

def quad(v,a):
    return scalar(mul(tr(v),mul(a,v)))

def main():

    cap=F(11,7)+delta
    eta=m/(m+cap)
    rho=1-eta
