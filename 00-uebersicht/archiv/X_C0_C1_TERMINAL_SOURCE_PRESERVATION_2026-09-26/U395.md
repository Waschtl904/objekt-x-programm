static uint16_t read_u16_be(FILE *f)
{
    unsigned char b[2];
    if (fread(b,1,2,f) != 2) die("truncated uint16");
    return ((uint16_t)b[0] << 8) | (uint16_t)b[1];
}

static uint32_t read_u32_be(FILE *f)
