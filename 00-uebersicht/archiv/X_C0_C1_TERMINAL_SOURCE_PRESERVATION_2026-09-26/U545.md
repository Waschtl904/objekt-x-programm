from fractions import Fraction as F

checks: list[str] = []

def zero(name: str, expr: s.Expr) -> None:
    value = s.factor(s.together(s.expand(expr)))
    if value != 0:
        raise AssertionError(f'{name}: residual {value}')
    checks.append(name)

def positive(name: str, value: F) -> None:

        raise AssertionError(f'{name}: {value} is not positive')

cy,sy=(Y+1/Y)/2,(Y-1/Y)/2
